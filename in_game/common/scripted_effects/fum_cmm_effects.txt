# Callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting, alert, or action bar element that was interacted with.
# Scope: country
fum_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# Common (common) Tab
		## Removals (removals) Group
		flag:fum__sop_removal = {
			fum_remove_sop_effect = yes
		}
		## Character Culling (character_culling) Group
		flag:fum__character_culling = {
			cmm_sync_bool_alias = {
				setting = fum__character_culling
				alias = fum__character_culling_setting
			}
		}
		flag:fum__max_characters_per_nation = {
			cmm_sync_setting_alias = {
				setting = fum__max_characters_per_nation
				alias = fum__max_characters_per_nation
			}
		}
		# World Reduction (world_reduction) Tab
		## General (general) Group
		flag:fum__reduction_preset = {
			fum_apply_reduction_preset = yes
		}
		flag:fum__apply_world_reduction = {
			fum_apply_world_reduction = yes
		}
		# QoL (qol) Tab
		## Speeds (speeds) Group
		flag:fum__army_movement_speed = {
			cmm_sync_setting_alias = {
				setting = fum__army_movement_speed
				alias = fum__army_movement_speed
			}
			fum_refresh_all_corrections = yes
		}
		flag:fum__navy_movement_speed = {
			cmm_sync_setting_alias = {
				setting = fum__navy_movement_speed
				alias = fum__navy_movement_speed
			}
			fum_refresh_all_corrections = yes
		}
		flag:fum__battle_length = {
			cmm_sync_dropdown_option_alias = {
				setting = fum__battle_length
				index = 1
				alias = fum__battle_length_vanilla
			}
			cmm_sync_dropdown_option_alias = {
				setting = fum__battle_length
				index = 2
				alias = fum__half_battle_length
			}
			cmm_sync_dropdown_option_alias = {
				setting = fum__battle_length
				index = 4
				alias = fum__double_battle_length
			}
			fum_refresh_all_corrections = yes
		}
	}
}

# Root scope: country. Called by every subcontinent list's _on_changed scripted GUI.
fum_reduction_list_changed = {
	cmm_apply_list_change = {
		setting = $SETTING$
	}
	# Option 1 is Custom.
	cmm_set_setting_value = {
		setting = fum__reduction_preset
		value = 1
	}
}

# Root scope: country. A preset ticks its subcontinents and clears everything else.
fum_apply_reduction_preset = {
	fum_clear_reduction_options = yes
	if = {
		limit = { "global_variable_map(cmm|flag:fum__reduction_preset)" = 2 } # Low Impact
		fum_tick_removal = { LIST = america ITEM = 1 } # North America
		fum_tick_removal = { LIST = america ITEM = 2 } # South America
		fum_tick_removal = { LIST = oceania ITEM = 1 } # Australasia
		fum_tick_removal = { LIST = oceania ITEM = 2 } # Pacific Islands
		fum_tick_removal = { LIST = africa ITEM = 3 } # Central Africa
		fum_tick_removal = { LIST = africa ITEM = 5 } # Southern Africa
		fum_tick_merge = { LIST = asia ITEM = 6 } # North Asia
		fum_tick_merge = { LIST = africa ITEM = 2 } # West Africa
		fum_tick_merge = { LIST = africa ITEM = 4 } # East Africa
	}
	else_if = {
		limit = { "global_variable_map(cmm|flag:fum__reduction_preset)" = 3 } # Europe Centric
		fum_tick_full_removal = { LIST = asia ITEM = 3 } # South Asia
		fum_tick_full_removal = { LIST = asia ITEM = 4 } # South East Asia
		fum_tick_full_removal = { LIST = asia ITEM = 5 } # East Asia
		fum_tick_pop_removal = { LIST = asia ITEM = 6 } # North Asia
		fum_tick_full_removal = { LIST = oceania ITEM = 1 } # Australasia
		fum_tick_full_removal = { LIST = oceania ITEM = 2 } # Pacific Islands
		fum_tick_removal = { LIST = america ITEM = 1 } # North America
		fum_tick_removal = { LIST = america ITEM = 2 } # South America
		fum_tick_removal = { LIST = africa ITEM = 3 } # Central Africa
		fum_tick_removal = { LIST = africa ITEM = 5 } # Southern Africa
		fum_tick_merge = { LIST = africa ITEM = 2 } # West Africa
		fum_tick_merge = { LIST = africa ITEM = 4 } # East Africa
	}
	else_if = {
		limit = { "global_variable_map(cmm|flag:fum__reduction_preset)" = 4 } # Asia Centric
		fum_tick_full_removal = { LIST = europe ITEM = 1 } # Western Europe
		fum_tick_full_removal = { LIST = europe ITEM = 2 } # Eastern Europe
		fum_tick_full_removal = { LIST = america ITEM = 1 } # North America
		fum_tick_full_removal = { LIST = america ITEM = 2 } # South America
		fum_tick_removal = { LIST = africa ITEM = 1 } # North Africa
		fum_tick_removal = { LIST = africa ITEM = 2 } # West Africa
		fum_tick_removal = { LIST = africa ITEM = 3 } # Central Africa
	}
}

# Root scope: country. Ticks Merge Pops.
fum_tick_merge = {
	cmm_set_list_field_value = { mod_id = fum setting_id = $LIST$ field_id = merge_pops item = $ITEM$ value = 1 }
}

# Root scope: country. Ticks Remove Countries and Merge Pops.
fum_tick_removal = {
	fum_tick_merge = { LIST = $LIST$ ITEM = $ITEM$ }
	cmm_set_list_field_value = { mod_id = fum setting_id = $LIST$ field_id = remove_countries item = $ITEM$ value = 1 }
}

# Root scope: country. Ticks Remove Countries, Merge Pops and Remove Pops.
fum_tick_pop_removal = {
	fum_tick_removal = { LIST = $LIST$ ITEM = $ITEM$ }
	cmm_set_list_field_value = { mod_id = fum setting_id = $LIST$ field_id = remove_pops item = $ITEM$ value = 1 }
}

# Root scope: country. Ticks every option.
fum_tick_full_removal = {
