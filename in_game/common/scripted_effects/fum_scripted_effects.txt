fum_remove_sop_effect = {
	every_country = {
		limit = {
			country_type = pop
		}
		change_country_type = building
		every_owned_building = {
			change_building_level = -10
		}
	}
}

# Root scope: country. Runs the ticked World Reduction options, one column at a time.
fum_apply_world_reduction = {
	clear_variable_list = fum_wr_remove_countries
	clear_variable_list = fum_wr_merge_pops
	clear_variable_list = fum_wr_remove_pops
	clear_variable_list = fum_wr_block_colonization
	fum_gather_reduction_columns = { SETTING = fum__america }
	fum_gather_reduction_columns = { SETTING = fum__oceania }
	fum_gather_reduction_columns = { SETTING = fum__asia }
	fum_gather_reduction_columns = { SETTING = fum__africa }
	fum_gather_reduction_columns = { SETTING = fum__europe }

	# Remove Pops includes the merge.
	every_in_list = {
		variable = fum_wr_remove_pops
		prev = {
			remove_list_variable = { name = fum_wr_merge_pops target = prev }
		}
	}

	every_in_list = {
		variable = fum_wr_remove_countries
		fum_remove_sub_continent_countries = yes
		set_global_variable = fum_countries_removed
	}
	every_in_list = {
		variable = fum_wr_merge_pops
		fum_merge_sub_continent_pops = yes
	}
	every_in_list = {
		variable = fum_wr_remove_pops
		fum_replace_sub_continent_pops = yes
	}
	every_in_list = {
		variable = fum_wr_block_colonization
		fum_block_sub_continent_colonization = yes
	}

	if = {
		limit = {
			is_target_in_variable_list = { name = fum_wr_remove_countries target = sub_continent:east_asia }
			is_target_in_variable_list = { name = fum_wr_remove_countries target = sub_continent:south_asia }
		}
		set_global_variable = fum_asia_removed
	}

	clear_variable_list = fum_wr_remove_countries
	clear_variable_list = fum_wr_merge_pops
	clear_variable_list = fum_wr_remove_pops
	clear_variable_list = fum_wr_block_colonization
}

# Root scope: country. Appends one subcontinent list's ticked rows to the four column lists.
fum_gather_reduction_columns = {
	fum_gather_reduction_column = { SETTING = $SETTING$ SLOT = 1 LIST = fum_wr_remove_countries }
	fum_gather_reduction_column = { SETTING = $SETTING$ SLOT = 2 LIST = fum_wr_merge_pops }
	fum_gather_reduction_column = { SETTING = $SETTING$ SLOT = 3 LIST = fum_wr_remove_pops }
	fum_gather_reduction_column = { SETTING = $SETTING$ SLOT = 4 LIST = fum_wr_block_colonization }
}

# Root scope: country.
fum_gather_reduction_column = {
	cmm_build_list_bool_list = { setting = $SETTING$ field_slot = $SLOT$ list_name = fum_wr_column }
	every_in_list = {
		variable = fum_wr_column
		prev = {
			add_to_variable_list = { name = $LIST$ target = prev }
		}
	}
	clear_variable_list = fum_wr_column
}

# Root scope: sub-continent.
fum_remove_sub_continent_countries = {
	every_region_in_continent = {
		limit = { fum_is_kept_region = no }
		fum_remove_region_countries = yes
	}
}

# Root scope: region.
fum_remove_region_countries = {
	every_location_in_region = {
		limit = { is_land = yes }
		fum_strip_location = yes
	}
}

# Root scope: location.
fum_strip_location = {
	every_unit_in_location = {
		destroy_unit = yes
	}
	every_buildings_in_location = {
		change_building_level = -100
	}
	if = {
		limit = {
			has_owner = yes
		}
		owner = {
			abandon_location = prev
		}
	}
}

# Root scope: sub-continent. Every region takes its largest culture and religion by population.
fum_merge_sub_continent_pops = {
	every_region_in_continent = {
		limit = { fum_is_kept_region = no }
		fum_merge_region_pops = yes
	}
}

# Root scope: region.
fum_merge_region_pops = {
	fum_find_region_majority = yes
	if = {
		limit = { local_var:fum_culture_best > 0 }
		fum_convert_region_countries = yes
		every_location_in_region = {
			limit = { is_land = yes }
			every_pop = {
				change_pop_culture = scope:fum_culture
				change_pop_religion = scope:fum_religion
			}
			force_refresh_culture_and_religion = this
		}
	}
}

# Root scope: sub-continent. Merges like fum_merge_sub_continent_pops, then leaves each location
# an undeveloped rural settlement of 10 people.
fum_replace_sub_continent_pops = {
	every_region_in_continent = {
		limit = { fum_is_kept_region = no }
		fum_replace_region_pops = yes
	}
}

# Root scope: region.
fum_replace_region_pops = {
	fum_find_region_majority = yes
	if = {
		limit = { local_var:fum_culture_best > 0 }
		fum_convert_region_countries = yes
		every_location_in_region = {
			limit = { is_land = yes }
			fum_replace_location_pops = { SIZE = 0.01 }
			change_development = -100
			change_location_rank = location_rank:rural_settlement
		}
	}
}

# Root scope: location. Leaves one pop of $SIZE$ of the culture and religion in scope:fum_culture and
# scope:fum_religion. The engine backfills an empty location from its own map-data culture and religion.
fum_replace_location_pops = {
	every_pop = {
		destroy_pop = this
	}
	add_pop = {
		culture = scope:fum_culture
		religion = scope:fum_religion
		type = pop_type:peasants
		size = $SIZE$
	}
	force_refresh_culture_and_religion = this
}

# Root scope: none. Console only: effect fum_repair_low_pop_locations
# Every land location under 10 people is left one 1k pop of its region's majority culture and
# religion.
fum_repair_low_pop_locations = {
	every_region = {
		fum_find_region_majority = yes
		if = {
			limit = { local_var:fum_culture_best > 0 }
			every_location_in_region = {
				limit = {
					is_land = yes
					population < 0.01
				}
				fum_replace_location_pops = { SIZE = 1 }
			}
		}
	}
}

# Root scope: region. Leaves the largest culture and religion by population in scope:fum_culture
# and scope:fum_religion. A region with no pops leaves local_var:fum_culture_best at 0.
fum_find_region_majority = {
	clear_local_variable_map = fum_culture_pop
	clear_local_variable_map = fum_religion_pop
	set_local_variable = { name = fum_culture_best value = 0 }
	set_local_variable = { name = fum_religion_best value = 0 }
	every_location_in_region = {
		every_pop = {
			fum_tally_pop_culture = yes
			fum_tally_pop_religion = yes
		}
	}
}

# Root scope: pop.
fum_tally_pop_culture = {
	culture = {
		save_scope_as = fum_tally_key
	}
	if = {
		limit = { is_key_in_local_variable_map = { name = fum_culture_pop target = scope:fum_tally_key } }
		set_local_variable = { name = fum_tally value = "local_variable_map(fum_culture_pop|scope:fum_tally_key)" }
	}
	else = {
		set_local_variable = { name = fum_tally value = 0 }
	}
	change_local_variable = { name = fum_tally add = pop_size }
	add_to_local_variable_map = { name = fum_culture_pop key = scope:fum_tally_key value = local_var:fum_tally }
	if = {
		limit = { local_var:fum_tally > local_var:fum_culture_best }
		set_local_variable = { name = fum_culture_best value = local_var:fum_tally }
		culture = {
			save_scope_as = fum_culture
		}
	}
}

# Root scope: pop.
fum_tally_pop_religion = {
	religion = {
		save_scope_as = fum_tally_key
	}
	if = {
		limit = { is_key_in_local_variable_map = { name = fum_religion_pop target = scope:fum_tally_key } }
		set_local_variable = { name = fum_tally value = "local_variable_map(fum_religion_pop|scope:fum_tally_key)" }
	}
	else = {
		set_local_variable = { name = fum_tally value = 0 }
	}
	change_local_variable = { name = fum_tally add = pop_size }
	add_to_local_variable_map = { name = fum_religion_pop key = scope:fum_tally_key value = local_var:fum_tally }
	if = {
		limit = { local_var:fum_tally > local_var:fum_religion_best }
		set_local_variable = { name = fum_religion_best value = local_var:fum_tally }
		religion = {
			save_scope_as = fum_religion
		}
	}
}

# Root scope: region.

	every_location_in_region = {
		limit = {
			is_capital = yes
			has_owner = yes
		}
		owner = {
			change_culture = scope:fum_culture
			change_religion = scope:fum_religion
			every_character = {
				change_character_culture = scope:fum_culture
				change_character_religion = scope:fum_religion
			}
		}
	}
}

# Root scope: region.
fum_block_region_colonization = {
	every_location_in_region = {
		limit = {
			is_land = yes
			NOT = { has_location_modifier = fum_location_disabled_static_modifier }
		}
		add_location_modifier = {
			modifier = fum_location_disabled_static_modifier
			years = -1
			mode = add
		}
	}
}

# Root scope: country. Culls the lowest-stat unprotected adults down to the max.
fum_cull_characters = {
	set_variable = { name = fum_to_cull value = 0 }
	every_character = {
		limit = { is_adult = yes }
		prev = { change_variable = { name = fum_to_cull add = 1 } }
	}
	change_variable = { name = fum_to_cull subtract = global_var:fum__max_characters_per_nation }

	if = {
		limit = { var:fum_to_cull > 0 }
		ordered_character = {
			order_by = {
				value = total_abilities
				multiply = -1
			}
			max = var:fum_to_cull
			check_range_bounds = no
			limit = { fum_is_protected_character = no }
			kill_character = this
		}
	}
	remove_variable = fum_to_cull
}

	fum_refresh_correction = { MODIFIER = combat_speed_modifier BASE = 1 DELTA = fum_battle_length_delta }
}

# Recomputes fum_corr_$MODIFIER$ as $DELTA$ x ($BASE$ + the country's other $MODIFIER$ sources) so
# the correction multiplies them instead of shifting them. $BASE$ is 1 where the engine applies the
# modifier as (1 + sum) and 0 where the sum is the whole quantity. modifier:$MODIFIER$ includes the
# correction already applied, which is whatever fum_corr_$MODIFIER$_value currently returns.
# Dropping the variable at a zero delta keeps a setting the player left off out of the save.
# Root scope: country.
fum_refresh_correction = {
	set_local_variable = {
		name = fum_delta
		value = $DELTA$
	}
	if = {
		limit = { local_var:fum_delta = 0 }
		if = {
			limit = { has_variable = fum_corr_$MODIFIER$ }
			remove_variable = fum_corr_$MODIFIER$
		}
	}
	else = {
		set_local_variable = {
			name = fum_others
			value = modifier:$MODIFIER$
		}
		change_local_variable = {
			name = fum_others
			subtract = fum_corr_$MODIFIER$_value
		}
		change_local_variable = {
			name = fum_others
			add = $BASE$
		}
		change_local_variable = {
			name = fum_others
			max = 0
		}
		set_variable = {
			name = fum_corr_$MODIFIER$
			value = {
				value = local_var:fum_others
				multiply = local_var:fum_delta
			}
		}
	}
}
