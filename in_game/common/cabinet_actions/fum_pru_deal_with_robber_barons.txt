# Stall timer for a vanilla issue where the action's date-based completion is missed. We do not cause this, but I was gaslit into believing we did thus I fixed the vanilla bug.
INJECT:pru_deal_with_robber_barons = {
	allow = {
		any_province = {
			has_province_modifier = PRU_robber_barons
		}
	}

	on_activate = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				set_variable = { name = fum_pru_robber_barons_months value = 0 }
			}
		}
	}

	on_deactivate = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				remove_variable = fum_pru_robber_barons_months
			}
		}
	}
}
