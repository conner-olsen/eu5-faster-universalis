# INJECT adds a per-target stall timer to vanilla's action. on_activate stamps a month counter on the target province; fum_iterate_robber_barons / fum_stop_robber_barons (monthly_country_pulse) increment it and force-remove PRU_robber_barons once it passes the threshold, covering the case where the engine's date-based completion is missed under FUM tick rates. Counter is per-province so several simultaneous actions each track their own target.
INJECT:pru_deal_with_robber_barons = {
	allow = {
		any_province = {
			has_province_modifier = PRU_robber_barons
		}
	}

	on_activate = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				set_variable = { name = fum_pru_robber_barons_months value = 0 }
			}
		}
	}

	on_deactivate = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				remove_variable = fum_pru_robber_barons_months
			}
		}
	}
}
