# Hook this mod into CMF shared registration on_action.
cmf_on_mod_registration = {
	on_actions = {
		fum_on_register_mod
	}
}

fum_on_register_mod = {
	effect = {
		fum_register_mod = yes
	}
}

# Unified callback hook for setting changes, alert clicks, action bar clicks.
cmf_on_callback = {
	on_actions = {
		fum_on_callback
	}
}

fum_on_callback = {
	effect = {
		fum_handle_callback = yes
	}
}
