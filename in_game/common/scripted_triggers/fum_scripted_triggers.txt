# Root scope: region.

	OR = {
		this = region:caucasus_region
		this = region:steppes_region
		this = region:egypt_region
	}
}

# Root scope: character. Returns yes if the character should not be culled.

	OR = {
		is_adult = no
		has_estate = estate_type:crown_estate
		in_cabinet = yes
		is_general = yes
		is_admiral = yes
		has_trait = architectural_visionary
	}
}
